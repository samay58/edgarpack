# Generalized Company Resolver Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Make `edgarpack identify`, `query`, `which`, `comps`, `compare`, `build`, `diff`, and `timeline` resolve any listed A-share or HKEX company by code or English name, without requiring per-company aliases in `universe.toml`.

**Architecture:** Two bundled JSON snapshots (`data/cn_stock_listings.json`, `data/hk_stock_listings.json`) loaded lazily into a `CompanyIndex` with O(1) code/alias lookups and a difflib fuzzy fallback. The CLI's existing `_resolve_cli_company` gains two new chain steps (snapshot lookup before SEC, live CNINFO/HKEX search after SEC) and an `--offline` opt-out. Successful live resolutions land in a 30-day on-disk cache under `~/.edgarpack/cache/identity/`. `universe.toml` semantics are unchanged; project-curated aliases still win.

**Tech Stack:** Python 3.11+, `httpx` (already in `china` extra), stdlib `difflib`, stdlib `json`, stdlib `pathlib`, stdlib `dataclasses`, pytest with the existing offline fixture pattern.

**Spec:** `docs/superpowers/specs/2026-04-27-generalized-company-resolver-design.md`

```python
"""English-name normalization for company alias matching."""

from __future__ import annotations

import re

_SUFFIX_PATTERN = re.compile(
    r",?\s*\b("
    r"co\.?(?:,?\s+ltd\.?)?"
    r"|corp(?:oration)?\.?"
    r"|inc\.?"
    r"|ltd\.?"
    r"|limited"
    r"|holdings?"
    r"|group"
    r"|plc"
    r"|sa"
    r"|ag"
    r")\b\.?\s*$",
    re.IGNORECASE,
)
_PUNCT_PATTERN = re.compile(r"[^\w\s]")
_WS_PATTERN = re.compile(r"\s+")


def normalize(value: str) -> str:
    """Return a canonical lowercase alias key for an English company name."""
    text = value.strip().lower()
    while True:
        replaced = _SUFFIX_PATTERN.sub("", text).strip().rstrip(",").strip()
        if replaced == text:
            break
        text = replaced
    text = _PUNCT_PATTERN.sub(" ", text)
    text = _WS_PATTERN.sub(" ", text).strip()
    return text
```

```python
"""Bundled SSE/SZSE/HKEX listings snapshot and lazy in-memory index."""

from __future__ import annotations

import difflib
import json
from dataclasses import dataclass, field
from pathlib import Path
from typing import Iterable, Literal

from .aliases import normalize

Source = Literal["SSE", "SZSE", "HKEX"]


@dataclass(frozen=True)
class CompanyRow:
    code: str
    source: Source
    market: str
    name_zh: str
    name_en: str
    aliases: tuple[str, ...] = field(default_factory=tuple)


def load_snapshot(path: Path) -> list[CompanyRow]:
    if not path.exists():
        return []
    payload = json.loads(path.read_text(encoding="utf-8"))
    rows = payload.get("rows", [])
    return [
        CompanyRow(
            code=str(r["code"]),
            source=r["source"],
            market=r.get("market", ""),
            name_zh=r.get("name_zh", ""),
            name_en=r.get("name_en", ""),
            aliases=tuple(r.get("aliases", [])),
        )
        for r in rows
    ]


def snapshot_date(path: Path) -> str | None:
    if not path.exists():
        return None
    payload = json.loads(path.read_text(encoding="utf-8"))
    return payload.get("snapshot_date")


def _canonical_code(code: str, source: str) -> str:
    if source == "HKEX":
        return code.lstrip("0").zfill(5)
    return code


def _key_variants_for_code(code: str) -> list[str]:
    stripped = code.lstrip("0")
    return list({code, stripped, stripped.zfill(5)})


class CompanyIndex:
    def __init__(self, rows: Iterable[CompanyRow]):
        self._by_code: dict[str, CompanyRow] = {}
        self._by_alias: dict[str, list[CompanyRow]] = {}
        self._all_aliases: list[str] = []

        for row in rows:
            canonical = _canonical_code(row.code, row.source)
            stored = CompanyRow(
                code=canonical,
                source=row.source,
                market=row.market,
                name_zh=row.name_zh,
                name_en=row.name_en,
                aliases=row.aliases,
            )
            for variant in _key_variants_for_code(canonical):
                self._by_code[variant] = stored

            keys: set[str] = set()
            if stored.name_en:
                keys.add(normalize(stored.name_en))
            for alias in stored.aliases:
                if alias:
                    keys.add(normalize(alias))
            for key in keys:
                if not key:
                    continue
                self._by_alias.setdefault(key, []).append(stored)

        self._all_aliases = sorted(self._by_alias.keys())

    @classmethod
    def load_paths(cls, paths: Iterable[Path]) -> "CompanyIndex":
        rows: list[CompanyRow] = []
        for path in paths:
            rows.extend(load_snapshot(path))
        return cls(rows)

    def lookup_by_code(self, code: str) -> CompanyRow | None:
        cleaned = code.strip()
        if not cleaned:
            return None
        return self._by_code.get(cleaned) or self._by_code.get(cleaned.zfill(5))

    def lookup_by_alias(self, value: str) -> list[CompanyRow]:
        key = normalize(value)
        if not key:
            return []
        return list(self._by_alias.get(key, []))

    def lookup_by_alias_fuzzy(self, value: str, *, cutoff: float = 0.85) -> list[CompanyRow]:
        key = normalize(value)
        if not key:
            return []
        matches = difflib.get_close_matches(key, self._all_aliases, n=3, cutoff=cutoff)
        if not matches:
            return []
        return list(self._by_alias.get(matches[0], []))
```

```python
"""Per-user disk cache for resolved company identities."""

from __future__ import annotations

import hashlib
import json
import time
from dataclasses import asdict
from pathlib import Path

from ..config import CACHE_DIR
from .aliases import normalize
from .snapshot import CompanyRow

DEFAULT_TTL_SECONDS = 30 * 86400


class IdentityCache:
    def __init__(self, root: Path | None = None, *, ttl_seconds: int = DEFAULT_TTL_SECONDS):
        base = root if root is not None else CACHE_DIR
        self.root = Path(base) / "identity"
        self.ttl = ttl_seconds
        self.root.mkdir(parents=True, exist_ok=True)

    def _path(self, key: str) -> Path:
        digest = hashlib.sha256(normalize(key).encode("utf-8")).hexdigest()
        return self.root / f"{digest}.json"

    def get(self, key: str) -> CompanyRow | None:
        path = self._path(key)
        if not path.exists():
            return None
        try:
            payload = json.loads(path.read_text(encoding="utf-8"))
        except (OSError, json.JSONDecodeError):
            return None
        resolved_at = float(payload.get("resolved_at", 0))
        if time.time() - resolved_at > self.ttl:
            return None
        row_data = payload.get("row")
        if not isinstance(row_data, dict):
            return None
        return CompanyRow(
            code=row_data["code"],
            source=row_data["source"],
            market=row_data.get("market", ""),
            name_zh=row_data.get("name_zh", ""),
            name_en=row_data.get("name_en", ""),
            aliases=tuple(row_data.get("aliases", [])),
        )

    def put(self, key: str, row: CompanyRow) -> None:
        payload = {
            "key": normalize(key),
            "resolved_at": time.time(),
            "row": asdict(row),
        }
        path = self._path(key)
        tmp = path.with_suffix(".tmp")
        tmp.write_text(json.dumps(payload), encoding="utf-8")
        tmp.replace(path)
```

```python
"""Live CNINFO + HKEX name-search fallback when bundled snapshots miss."""

from __future__ import annotations

from typing import Any

from .aliases import normalize
from .snapshot import CompanyRow


def _fetch_cninfo(query: str, *, timeout: float) -> dict[str, Any]:
    import httpx

    headers = {
        "Accept": "application/json, text/javascript, */*; q=0.01",
        "Content-Type": "application/x-www-form-urlencoded; charset=UTF-8",
        "Origin": "https://www.cninfo.com.cn",
        "Referer": "https://www.cninfo.com.cn/new/commonUrl/pageOfSearch",
        "User-Agent": "edgarpack/0.1 (+https://github.com)",
    }
    data = {
        "pageNum": "1",
        "pageSize": "10",
        "tabName": "fulltext",
        "searchkey": query,
        "category": "category_ndbg_szsh",
        "isHLtitle": "true",
    }
    with httpx.Client(timeout=timeout, follow_redirects=True) as client:
        response = client.post(
            "https://www.cninfo.com.cn/new/hisAnnouncement/query",
            headers=headers,
            data=data,
        )
        response.raise_for_status()
        return response.json()


def _fetch_hkex(query: str, *, timeout: float) -> dict[str, Any]:
    import httpx

    headers = {
        "Accept": "application/json",
        "User-Agent": "edgarpack/0.1 (+https://github.com)",
    }
    params = {"sym": query, "lang": "EN"}
    url = "https://www1.hkex.com.hk/hkexwidget/data/getequityquote"
    with httpx.Client(timeout=timeout, follow_redirects=True) as client:
        response = client.get(url, params=params, headers=headers)
        response.raise_for_status()
        return response.json()


def _parse_cninfo_search(payload: dict[str, Any], *, query: str) -> list[CompanyRow]:
    announcements = payload.get("announcements") or []
    if not isinstance(announcements, list):
        return []
    seen: set[str] = set()
    rows: list[CompanyRow] = []
    for item in announcements:
        if not isinstance(item, dict):
            continue
        code = str(item.get("secCode") or "").strip()
        if not code or code in seen:
            continue
        seen.add(code)
        name_zh = str(item.get("secName") or "").strip()
        name_en = str(item.get("orgName") or item.get("companyName") or "").strip()
        source = "SSE" if code.startswith("6") else "SZSE"
        rows.append(
            CompanyRow(
                code=code,
                source=source,
                market="",
                name_zh=name_zh,
                name_en=name_en,
                aliases=(normalize(name_en),) if name_en else (),
            )
        )
    return rows


def _parse_hkex_search(payload: dict[str, Any]) -> list[CompanyRow]:
    data = payload.get("data") or {}
    stocks = data.get("stockList") or []
    rows: list[CompanyRow] = []
    for item in stocks:
        if not isinstance(item, dict):
            continue
        sym = str(item.get("sym") or "").strip()
        if not sym:
            continue
        code = sym.zfill(5)
        name_en = str(item.get("name") or "").strip()
        name_zh = str(item.get("nameCn") or "").strip()
        rows.append(
            CompanyRow(
                code=code,
                source="HKEX",
                market="",
                name_zh=name_zh,
                name_en=name_en,
                aliases=(normalize(name_en),) if name_en else (),
            )
        )
    return rows


def search_live(query: str, *, timeout: float = 5.0) -> CompanyRow | None:
    """Hit CNINFO and HKEX search; return the first matching CompanyRow or None."""
    target = normalize(query)
    if not target:
        return None
    try:
        cninfo_payload = _fetch_cninfo(query, timeout=timeout)
    except Exception:
        cninfo_payload = {"announcements": []}
    cn_rows = _parse_cninfo_search(cninfo_payload, query=query)
    if cn_rows:
        return cn_rows[0]
    try:
        hkex_payload = _fetch_hkex(query, timeout=timeout)
    except Exception:
        hkex_payload = {"data": {"stockList": []}}
    hk_rows = _parse_hkex_search(hkex_payload)
    if hk_rows:
        return hk_rows[0]
    return None
```

```python
from __future__ import annotations

from pathlib import Path

from ..errors import AmbiguousCompany
from .cache import IdentityCache
from .live import search_live
from .snapshot import CompanyIndex, CompanyRow, bundled_paths, load_snapshot, snapshot_date

__all__ = [
    "AmbiguousCompany",
    "CompanyIndex",
    "CompanyRow",
    "bundled_paths",
    "load_snapshot",
    "snapshot_date",
    "search_live",
    "resolve_external",
]

def _default_index() -> CompanyIndex:
    global _DEFAULT_INDEX
    if _DEFAULT_INDEX is None:
        _DEFAULT_INDEX = CompanyIndex.load_paths(bundled_paths())
    return _DEFAULT_INDEX


def resolve_external(
    query: str,
    *,
    index: CompanyIndex | None = None,
    cache_root: Path | None = None,
    offline: bool = False,
    timeout: float = 5.0,
) -> CompanyRow | None:
    """Resolve a query against the bundled snapshot, then optionally a live search.

    Returns None if nothing matched. Raises AmbiguousCompany when the snapshot
    yields more than one candidate from a cross-listed alias.
    """
    if not query or not query.strip():
        return None

    idx = index if index is not None else _default_index()
    cleaned = query.strip()

    by_code = idx.lookup_by_code(cleaned)
    if by_code is not None:
        return by_code

    exact = idx.lookup_by_alias(cleaned)
    if len(exact) == 1:
        return exact[0]
    if len(exact) > 1:
        rendered = ", ".join(f"{r.code} ({r.source})" for r in exact)
        raise AmbiguousCompany(
            f"{query!r} resolves to multiple listings: {rendered}. "
            "Pass the stock code to disambiguate."
        )

    fuzzy = idx.lookup_by_alias_fuzzy(cleaned)
    if len(fuzzy) == 1:
        return fuzzy[0]
    if len(fuzzy) > 1:
        rendered = ", ".join(f"{r.code} ({r.source})" for r in fuzzy)
        raise AmbiguousCompany(
            f"{query!r} resolves to multiple listings: {rendered}. "
            "Pass the stock code to disambiguate."
        )

    if offline:
        return None

    cache = IdentityCache(cache_root)
    cached = cache.get(query)
    if cached is not None:
        return cached

    live = search_live(query, timeout=timeout)
    if live is not None:
        cache.put(query, live)
    return live
```

```python
#!/usr/bin/env python3
"""Regenerate data/cn_stock_listings.json + data/hk_stock_listings.json.

Pulls the public CNINFO stock-list endpoint (SSE main + STAR + SZSE main + ChiNext)
and the HKEX securities list, normalizes each row, generates aliases, and writes
both JSON files with today's snapshot_date.

Run on demand. Network only; no auth required.
"""

from __future__ import annotations

import datetime as dt
import json
import sys
from pathlib import Path
from typing import Any

import httpx

REPO_ROOT = Path(__file__).resolve().parent.parent
DATA_DIR = REPO_ROOT / "data"

CNINFO_STOCK_LIST = "http://www.cninfo.com.cn/new/data/szse_stock.json"
SSE_STOCK_LIST = "http://www.cninfo.com.cn/new/data/sse_stock.json"
HKEX_STOCK_LIST = (
    "https://www1.hkex.com.hk/hkexwidget/data/getequityfilter"
    "?lang=eng&token=&qid=&callback=&sort=5&order=0&all=1"
)


def _fetch_json(url: str, *, timeout: float = 30.0) -> dict[str, Any]:
    with httpx.Client(
        timeout=timeout,
        follow_redirects=True,
        headers={"User-Agent": "edgarpack-refresh/0.1"},
    ) as client:
        response = client.get(url)
        response.raise_for_status()
        text = response.text
        if text.startswith("(") and text.endswith(")"):
            text = text[1:-1]
        return json.loads(text)


def _market_for_a_share(code: str) -> tuple[str, str]:
    if code.startswith("688"):
        return "SSE", "STAR"
    if code.startswith(("60", "601", "603", "605")):
        return "SSE", "Main"
    if code.startswith("300") or code.startswith("301"):
        return "SZSE", "ChiNext"
    return "SZSE", "Main"


def _normalize_alias(value: str) -> str:
    from edgarpack.resolve.aliases import normalize

    return normalize(value)


def _generate_aliases(name_en: str) -> list[str]:
    if not name_en:
        return []
    seen = {_normalize_alias(name_en)}
    seen.discard("")
    return sorted(seen)


def fetch_a_share_rows() -> list[dict[str, Any]]:
    rows: list[dict[str, Any]] = []
    for url in (CNINFO_STOCK_LIST, SSE_STOCK_LIST):
        payload = _fetch_json(url)
        items = payload.get("stockList") or payload.get("data") or []
        for item in items:
            code = str(item.get("code") or item.get("zqdm") or "").strip()
            if not code:
                continue
            source, market = _market_for_a_share(code)
            name_zh = str(item.get("zwjc") or item.get("name") or "").strip()
            name_en = str(item.get("ywjc") or item.get("nameEn") or "").strip()
            rows.append(
                {
                    "code": code,
                    "source": source,
                    "market": market,
                    "name_zh": name_zh,
                    "name_en": name_en,
                    "aliases": _generate_aliases(name_en),
                }
            )
    return rows


def fetch_hkex_rows() -> list[dict[str, Any]]:
    payload = _fetch_json(HKEX_STOCK_LIST)
    data = payload.get("data") or {}
    stocks = data.get("stockList") or []
    rows: list[dict[str, Any]] = []
    for item in stocks:
        sym = str(item.get("sym") or "").strip()
        if not sym:
            continue
        code = sym.zfill(5)
        name_en = str(item.get("nm") or item.get("name") or "").strip()
        name_zh = str(item.get("nameCn") or "").strip()
        rows.append(
            {
                "code": code,
                "source": "HKEX",
                "market": "Main",
                "name_zh": name_zh,
                "name_en": name_en,
                "aliases": _generate_aliases(name_en),
            }
        )
    return rows


def write_snapshot(path: Path, rows: list[dict[str, Any]]) -> None:
    payload = {"snapshot_date": dt.date.today().isoformat(), "rows": rows}
    path.write_text(json.dumps(payload, ensure_ascii=False, indent=2), encoding="utf-8")


def main() -> int:
    DATA_DIR.mkdir(exist_ok=True)
    cn_rows = fetch_a_share_rows()
    hk_rows = fetch_hkex_rows()
    if not cn_rows:
        print("ERROR: CNINFO returned zero rows; refusing to write empty snapshot.", file=sys.stderr)
        return 1
    if not hk_rows:
        print("ERROR: HKEX returned zero rows; refusing to write empty snapshot.", file=sys.stderr)
        return 1
    write_snapshot(DATA_DIR / "cn_stock_listings.json", cn_rows)
    write_snapshot(DATA_DIR / "hk_stock_listings.json", hk_rows)
    print(f"Wrote {len(cn_rows)} A-share rows and {len(hk_rows)} HKEX rows.")
    return 0


if __name__ == "__main__":
    raise SystemExit(main())
```
